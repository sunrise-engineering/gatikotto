      <!-- Calculations -->
      <h2 class="mt-5">Common Mechanical Engineering Calculations</h2>

      <!-- Force -->
      <div class="card shadow-sm mb-4">
        <div class="card-body">
          <h5 class="card-title text-primary">1. Force Calculation</h5>
          <p class="mb-2"><strong>Formula:</strong></p>
          <div class="bg-light p-3 text-center rounded mb-2">
            <code>F = m × a</code>
          </div>
          <p class="small text-muted mb-0">
            F = Force (Newton), m = Mass (kg), a = Acceleration (m/s²)
          </p>
        </div>
      </div>

      <!-- Power -->
      <div class="card shadow-sm mb-4">
        <div class="card-body">
          <h5 class="card-title text-success">2. Power Calculation</h5>
          <p class="mb-2"><strong>Formula:</strong></p>
          <div class="bg-light p-3 text-center rounded mb-2">
            <code>P = W / t</code>
          </div>
          <p class="small text-muted mb-0">
            P = Power (Watts), W = Work (Joules), t = Time (seconds)
          </p>
        </div>
      </div>

      <!-- Efficiency -->
      <div class="card shadow-sm mb-4">
        <div class="card-body">
          <h5 class="card-title text-warning">3. Efficiency Calculation</h5>
          <p class="mb-2"><strong>Formula:</strong></p>
          <div class="bg-light p-3 text-center rounded mb-2">
            <code>Efficiency (%) = (Output / Input) × 100</code>
          </div>
        </div>
      </div>

      <!-- Stress -->
      <div class="card shadow-sm mb-4">
        <div class="card-body">
          <h5 class="card-title text-danger">4. Stress Calculation</h5>
          <p class="mb-2"><strong>Formula:</strong></p>
          <div class="bg-light p-3 text-center rounded mb-2">
            <code>Stress = Force / Area</code>
          </div>
        </div>
      </div>
